% Figure: the model network, and the LoRA adapter on one weight inside a block.
% Facts come from the published model repository and model card.
% Requires in the preamble: \usepackage{tikz} + \usetikzlibrary{positioning,arrows.meta},
%                            and amsmath for \frac in the adapter formula.
\begin{figure}[t]
  \centering
  \begin{tikzpicture}[
    x=1cm, y=1cm,
    font=\small,
    block/.style={rectangle, rounded corners=2pt, draw=black!70, line width=0.5pt,
      fill=blue!10, align=center, inner sep=4pt, minimum height=9mm},
    inputblock/.style={block, fill=green!12},
    stageblock/.style={block, fill=orange!15},
    noteblock/.style={rectangle, rounded corners=2pt, draw=black!45, dashed,
      fill=red!8, align=left, inner sep=3pt, font=\scriptsize},
    flow/.style={->, >=Latex, line width=0.7pt, draw=black!75},
    feed/.style={->, >=Latex, line width=0.7pt, draw=black!75, dashed},
    arrowlab/.style={font=\scriptsize, fill=white, inner sep=1pt},
  ]

  % ================= left part: the path of the model =================
  \node[inputblock, anchor=north, text width=5.60cm, minimum height=0.80cm] (lat) at (5.62,14.60)
    {\textbf{latent token}, 128 channels};
  \node[block, text width=5.60cm, minimum height=0.80cm, below=0.28cm of lat] (proj)
    {\textbf{project to 3072}};
  \node[block, text width=5.60cm, minimum height=1.10cm, below=0.28cm of proj] (ds)
    {\textbf{5 double-stream blocks}\\[1pt]\scriptsize text and image tokens stay apart,
     joint attention};
  \node[block, text width=5.60cm, minimum height=0.80cm, below=0.28cm of ds] (cat)
    {\textbf{concatenate the two streams}};
  \node[block, text width=5.60cm, minimum height=1.10cm, below=0.28cm of cat] (ss)
    {\textbf{20 single-stream blocks}\\[1pt]\scriptsize one stream, joint attention};
  \node[block, text width=5.60cm, minimum height=1.10cm, below=0.28cm of ss] (fin)
    {\textbf{final layer}\\[1pt]\scriptsize adaptive layer norm, project to 128 channels};
  \node[inputblock, text width=5.60cm, minimum height=1.10cm, below=0.28cm of fin] (latout)
    {\textbf{predicted direction}\\[1pt]\scriptsize the velocity for one update};
  \node[stageblock, text width=5.60cm, minimum height=1.10cm, below=0.28cm of latout] (upd)
    {\textbf{scheduler / Euler update}\\[1pt]\scriptsize the sampler moves the latent along the direction};
  \node[inputblock, text width=5.60cm, minimum height=1.00cm, below=0.28cm of upd] (latupd)
    {\textbf{updated latent}\\[1pt]\scriptsize the next input, four steps in all};
  \node[block, text width=5.60cm, minimum height=1.00cm, below=0.28cm of latupd] (dec)
    {\textbf{VAE decode}\\[1pt]\scriptsize the final latent only};
  \foreach \a/\b in {lat/proj, proj/ds, ds/cat, cat/ss, ss/fin, fin/latout, latout/upd, upd/latupd, latupd/dec}
    {\draw[flow] (\a.south) -- (\b.north);}

  % the update loop: the updated latent becomes the next input, and only the last one is decoded
  \draw[feed] (latupd.east) -- ++(0.35,0) |- (latout.east);
  \node[arrowlab, anchor=west] at ([xshift=0.50cm]latupd.east) {next step, four in all};

  % the text encoder enters through joint attention (dashed = conditioning path)
  \node[block, text width=1.70cm, left=0.40cm of cat] (enc)
    {\textbf{Qwen3-4B}\\[1pt]\scriptsize text encoder};
  \coordinate (txtcorner) at (enc.north |- ds.center);
  \draw[feed] (enc.north) -- (txtcorner) -- (ds.west);
  \path (txtcorner) -- (ds.west) coordinate[midway] (txtmid);
  \node[arrowlab, align=center, text width=1.70cm, anchor=south] at ([yshift=0.14cm]txtmid)
    {text tokens,\\width 7680};

  % the timestep embedding feeds both block rows and the final layer
  \path (ds.east) -- ++(0.55,0) coordinate (tbus);
  \draw[draw=black!75, line width=0.7pt, dashed] (tbus |- ds.center) -- (tbus |- fin.south);
  \foreach \s in {ds,ss,fin}{\draw[feed] (tbus |- \s.center) -- (\s.east);}
  \node[block, text width=2.10cm, minimum height=1.20cm, anchor=west] (tstep)
    at ([xshift=0.15cm]tbus |- fin.south)
    {\textbf{timestep embedding}\\[1pt]\scriptsize sets scale and shift for every block};
  \draw[feed] (tbus |- fin.south) -- (tstep.west);

  % ================= right part: the adapter zoom, linked to the stack =================
  \node[inputblock, text width=2.00cm, minimum height=0.75cm] (layin)
    at ([xshift=4.54cm]ds.north east) {layer input $x$};

  % a bracket spans the two block rows and runs to the zoom
  \coordinate (brT) at ([xshift=0.95cm]ds.north east);
  \coordinate (brB) at ([xshift=0.95cm]ss.south east);
  \draw[draw=black!75, line width=0.7pt] (brT) -- (brB);
  \draw[draw=black!75, line width=0.7pt] ([xshift=-0.18cm]brT) -- (brT);
  \draw[draw=black!75, line width=0.7pt] (brB) -- ([xshift=-0.18cm]brB);
  \draw[flow] (brT) -- (layin.west);
  \path (brT) -- (layin.west) coordinate[midway] (linkmid);
  \node[arrowlab, align=center, text width=1.80cm, anchor=south]
    at ([yshift=0.12cm]linkmid) {one of 80\\adapted weights};

  \coordinate (zbr) at ([yshift=-0.70cm]layin.south);
  \node[block, text width=2.15cm, minimum height=2.05cm] (wgt)
    at ([xshift=-0.85cm,yshift=-1.975cm]layin.south)
    {\textbf{$W$}\\[1pt]\scriptsize the block's own\\[1pt]weight, frozen};
  \node[block, text width=0.45cm, minimum height=0.90cm] (bA)
    at ([xshift=0.70cm,yshift=-1.40cm]layin.south) {\textbf{A}};
  \node[block, text width=0.45cm, minimum height=0.90cm, below=0.20cm of bA] (bB) {\textbf{B}};
  \draw[flow] (bA.south) -- (bB.north);
  \draw[draw=black!75, line width=0.7pt] (layin.south) -- (zbr);
  \draw[draw=black!75, line width=0.7pt] (zbr -| wgt.north) -- (zbr -| bA.north);
  \draw[flow] (zbr -| wgt.north) -- (wgt.north);
  \draw[flow] (zbr -| bA.north) -- (bA.north);

  \coordinate (zmg) at ([yshift=-0.60cm]bB.south);
  \draw[flow] (wgt.south) -- (zmg -| wgt.south);
  \draw[flow] (bB.south) -- (zmg -| bB.south);
  \draw[draw=black!75, line width=0.7pt] (zmg -| wgt.south) -- (zmg -| bB.south);
  \node[circle, draw=black!70, line width=0.5pt, fill=white, inner sep=0.6pt] at (zmg) {$+$};
  \node[inputblock, text width=1.85cm, minimum height=0.75cm] (zout)
    at ([yshift=-0.725cm]zmg) {output};
  \draw[flow] (zmg) -- (zout.north);
  \node[font=\small] (formula) at ([yshift=-0.55cm]zout.south) {$Wx + \frac{\alpha}{r}BAx$};

  \node[font=\scriptsize, align=left, text width=1.20cm, anchor=west] (abl)
    at ([xshift=0.10cm]bA.east) {\textbf{adapter}\\[1pt]rank 128,\\trained};

  % ================= notes =================
  \coordinate (notetop) at ([yshift=-0.35cm]dec.south);
  \node[noteblock, text width=2.10cm, anchor=north west] at (0.45,0 |- notetop)
    {This layer stack is the published layout. We did not change it.};
  \node[noteblock, text width=4.40cm, anchor=north west] at (4.425,0 |- notetop)
    {Only A and B are trained. $W$ never changes, so the adapter is small and it can be
     removed again.};
  \node[noteblock, text width=4.40cm, anchor=north west] at (10.70,0 |- notetop)
    {The adapter adapts 80 weights. In each double-stream block it takes the
     attention projections and the feed-forward linears of both streams, eight
     weights in all. In each single-stream block it takes the two fused linears.};

  \end{tikzpicture}
  \caption{The model path: the latent token is projected to width 3072, passes the
    double-stream and single-stream blocks, and the final layer predicts a direction, the
    velocity for one update. The sampler applies the update, and only the final latent
    reaches the VAE decode. The text encoder and the timestep embedding enter as
    conditioning. The bracket on the right marks one weight inside a block, where a small
    trained adapter sits beside the block's own frozen weight.}
  \label{fig:architecture_lora}
\end{figure}
