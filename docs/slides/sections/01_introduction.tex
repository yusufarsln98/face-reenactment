\section{Introduction}

\begin{frame}{Problem: Face Reenactment}
    \begin{columns}[c]
        \begin{column}{0.55\textwidth}
            \textbf{Goal}
            \begin{itemize}
                \item Animate a static \textbf{Source ($S$)} photo using motion from a \textbf{Driving ($D$)} video.
                \item Keep identity of $S$, transfer motion from $D$.
            \end{itemize}

            \vspace{0.3cm}
            \textbf{Key Challenges}
            \begin{itemize}
                \item \textbf{Identity preservation:} Avoid warping face features into driving face.
                \item \textbf{Non-rigid motion:} Facial skin stretches and deforms (not just rigid 2D shift).
                \item \textbf{Occlusions:} Mouth opening reveals unseen areas (teeth, tongue).
            \end{itemize}
        \end{column}

        \begin{column}{0.44\textwidth}
            \begin{block}{Input / Output}
                \small
                \textbf{Inputs:}
                \begin{itemize}
                    \item One static portrait photo ($S$)
                    \item A driving video sequence ($D$)
                \end{itemize}
                \vspace{0.15cm}
                \textbf{Output:}
                \begin{itemize}
                    \item Video of the person in $S$
                    \item Mimicking the expressions and head movements of $D$
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
