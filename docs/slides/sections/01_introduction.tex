\section{Introduction}

\begin{frame}{Problem Formulation: Face Reenactment}
    \begin{columns}[c]
        \begin{column}{0.54\textwidth}
            \small
            \textbf{Core Objective}
            \begin{itemize}
                \item Animate static \textbf{Source ($S$)} photo using motion from \textbf{Driving ($D$)} video.
                \item Preserve identity of $S$, transfer motion dynamics from $D$.
            \end{itemize}

            \vspace{0.15cm}
            \textbf{Key Technical Challenges}
            \begin{itemize}
                \item \textbf{Identity preservation:} Avoid warping face features into driving face.
                \item \textbf{Non-rigid deformation:} Facial skin stretches (not just rigid 2D shift).
                \item \textbf{Occlusion handling:} Mouth opening reveals unseen areas (teeth, tongue).
            \end{itemize}
        \end{column}

        \begin{column}{0.44\textwidth}
            \begin{block}{Input / Output Setup}
                \small
                \textbf{Inputs:}
                \begin{itemize}
                    \item One static portrait photo ($S$)
                    \item A driving video sequence ($D$)
                \end{itemize}
                \vspace{0.1cm}
                \textbf{Output:}
                \begin{itemize}
                    \item Reenacted video of $S$
                    \item Mimicking expressions and head pose of $D$
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
