\section{Method 2: GANs \& FOMM}

% --- Slide 1: Standard GAN ---
\begin{frame}{Standard GAN Architecture}
    \begin{columns}[c]
        \begin{column}{0.48\textwidth}
            \small
            \textbf{How GANs Work (Goodfellow et al., 2014)}
            \begin{itemize}
                \item Two neural networks competing in an adversarial setup:
                \item \textbf{Generator ($G$):} Takes noise $\mathbf{z}$ and learns to generate realistic face images.
                \item \textbf{Discriminator ($D$):} Evaluates images and decides if real or fake.
                \item \textbf{Adversarial training:} $G$ gets feedback from $D$ and learns photorealistic textures.
            \end{itemize}
        \end{column}

\end{frame}

% --- Slide 2: Why Standard GANs Fail ---
\begin{frame}{The Controllability Issue in Vanilla GANs}
    \begin{columns}[c]
        \begin{column}{0.50\textwidth}
            \begin{block}{Why Unconditional GANs Fall Short}
                \small
                \begin{itemize}
                    \item Generates arbitrary faces from noise $\mathbf{z}$.
                    \item Identity, pose, and emotion are entangled in $\mathbf{z}$.
                    \item Cannot say: \textit{"keep Person A, copy Person B's expression"}.
                \end{itemize}
            \end{block}
            
            \vspace{-0.1cm}
            \begin{block}{What Face Reenactment Requires}
                \small
                \begin{itemize}
                    \item Keep source identity fixed: appearance from $S$.
                    \item Motion driven externally: dynamics from $D$.
                \end{itemize}
            \end{block}
        \end{column}
