\section{Method 2: Keypoint-Based FOMM}

% --- Slide 1: FOMM Overview ---
\begin{frame}{First Order Motion Model (Siarohin et al., NeurIPS 2019)}
    \begin{columns}[c]
        \begin{column}{0.54\textwidth}
            \small
            \textbf{Core Innovations}
            \begin{itemize}
                \item \textbf{No Pretrained Landmarks:} Doesn't need 68 landmark detectors or 3D models (3DMM).
                \item \textbf{Self-Supervised:} Discovers keypoints automatically from raw video pairs.
                \item \textbf{Decoupled Repr.:} Appearance from source $S$; motion from driving $D$.
                \item \textbf{Domain Agnostic:} Works on human faces, bodies (Tai-Chi), and animals.
            \end{itemize}

            \vspace{0.12cm}
            \footnotesize \textbf{No adversarial loss:} this project trains FOMM without a discriminator.
        \end{column}
