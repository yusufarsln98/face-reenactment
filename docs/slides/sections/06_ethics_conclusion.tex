% --- Slide 1: Ethics & Detection ---
\begin{frame}{Ethical Implications \& DeepFake Detection}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Societal Risks}
                \small
                \begin{itemize}
                    \item \textbf{Unauthorized Impersonation:} Reenacting individuals from a single photo without consent.
                    \item \textbf{Misinformation:} Generating fabricated audiovisual material to deceive public opinion.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Detection Strategies}
                \small
                \begin{itemize}
                    \item \textbf{Biological signals:} Unnatural blinking dynamics and subtle facial blood-flow pulses (rPPG).
                    \item \textbf{Frequency artifacts:} High-frequency spectral artifacts from neural upsampling layers.
                    \item \textbf{Watermarking / C2PA:} Cryptographic metadata and provenance tracking embedded at capture.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Conclusion}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \textbf{Summary}
            \begin{itemize}
                \item FOMM animates single portraits using 10 unsupervised keypoints and local Jacobian transformations.
                \item Eliminates dependency on manual facial landmarks or pre-built 3D face models.
                \item Inpainting handles disocclusions, but rare traits (e.g., dental braces) vanish due to dataset bias and lack of temporal memory.
            \end{itemize}
        \end{column}
