\section{Ethics \& Conclusion}

% --- Slide 1: Ethics & Detection ---
\begin{frame}{Ethics \& DeepFake Detection}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Risks}
                \small
                \begin{itemize}
                    \item \textbf{Impersonation:} Reenacting individuals without their knowledge or consent.
                    \item \textbf{Misinformation:} Generating fabricated video statements.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Detection Methods}
                \small
                \begin{itemize}
                    \item \textbf{Biological signals:} Heartbeat blood flow (rPPG), unnatural blinking patterns.
                    \item \textbf{Frequency artifacts:} High-frequency noise left by GAN upsampling layers.
                    \item \textbf{Watermarking / C2PA:} Cryptographic metadata embedded in original camera recordings.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Conclusion}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \textbf{Summary}
            \begin{itemize}
                \item FOMM successfully animates single portraits using 10 unsupervised keypoints and Jacobians.
                \item Eliminates need for landmark detectors or 3D face models.
                \item Self-occlusions are handled by inpainting, but rare features (braces) vanish due to lack of temporal memory.
            \end{itemize}
        \end{column}
