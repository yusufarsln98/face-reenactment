\begin{frame}{Ethical Implications \& DeepFake Detection}
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Societal Risks}
                \small
                \begin{itemize}
                    \item \textbf{Impersonation:} Reenacting individuals without their consent.
                    \item \textbf{Misinformation:} Generating fabricated videos to deceive the public.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Detection Strategies}
                \small
                \begin{itemize}
                    \item \textbf{Biological signals:} Unnatural blinking and pulse signals (rPPG).
                    \item \textbf{Frequency artifacts:} High-frequency noise from neural upsampling.
                    \item \textbf{Watermarking / C2PA:} Cryptographic metadata embedded at capture.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
