\begin{frame}{Self-Occlusion \& Inpainting Mechanism}
    \begin{columns}[c]
        \begin{column}{0.50\textwidth}
            \begin{block}{The Occlusion Problem}
                \small
                \begin{itemize}
                    \item Turning head or opening mouth reveals unseen areas in $S$ (teeth, tongue).
                    \item Pure warping has no source pixels $\to$ causes stretching or black holes.
                \end{itemize}
            \end{block}

            \vspace{-0.1cm}
            \begin{block}{The FOMM Solution}
                \small
                \begin{itemize}
                    \item \textbf{Occlusion Mask ($O$):} Flags missing pixels ($O \approx 0$).
                    \item \textbf{Inpainting:} Generator fills missing pixels using learned facial priors.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \centering
            \includegraphics[width=\linewidth, height=3.8cm, keepaspectratio]{diagram_occlusion_inpainting.png}
        \end{column}
    \end{columns}
\end{frame}
