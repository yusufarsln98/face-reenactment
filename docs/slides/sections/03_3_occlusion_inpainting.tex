\begin{frame}{Self-Occlusion \& Inpainting Mechanism}
    \begin{columns}[c]
        \begin{column}{0.48\textwidth}
            \begin{block}{The Occlusion Problem}
                \small
                \begin{itemize}
                    \item Head rotations or mouth opening reveal regions hidden in source portrait $S$ (teeth, tongue, cheeks).
                    \item Pure warping has no source pixels to sample from $\to$ results in severe stretching or black holes.
                \end{itemize}
            \end{block}

            \begin{block}{The FOMM Solution}
                \small
                \begin{itemize}
                    \item \textbf{Occlusion Mask ($O$):} Detects missing or disoccluded pixels ($O \approx 0$).
                    \item \textbf{Inpainting:} Generator fills missing pixels using facial patterns learned during training.
                \end{itemize}
            \end{block}
        \end{column}

    \end{columns}
\end{frame}
