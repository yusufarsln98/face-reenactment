% --- Slide 1: Hypothesis & Setup for Edge Case ---
\begin{frame}{Failure Mode Experiment: Dental Braces}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{block}{Experimental Setup}
                \small
                \begin{itemize}
                    \item \textbf{Motivation:} Test how FOMM handles rare, out-of-distribution (OOD) facial features not present in training data.
                    \item \textbf{Subject:} Source portrait with metallic dental braces (\texttt{bella\_poarch\_dis\_teli}).
                    \item \textbf{Driving Motion:} Trajectory moving from head tilt to lip closure, ending in a wide squinting smile.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Hypothesis \& Theoretical Basis}
                \small
                \begin{itemize}
                    \item VoxCeleb training distribution contains virtually zero faces with dental braces.
                    \item \textbf{Warping Regime ($O \approx 1$):} Optical flow directly samples source pixels $\to$ braces will be preserved.
                    \item \textbf{Inpainting Regime ($O \approx 0$):} Under large deformation, model marks teeth as occluded and synthesizes them.
                    \item \textbf{Prediction:} Inpainting will sample dataset average and erase braces.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- Slide 3: Root Cause Analysis & Video Links ---
\begin{frame}{Technical Mechanism: Dual-Path Synthesis}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \begin{block}{Synthesis Formulation}
                \scriptsize
                $$\text{Output} = (\text{Warped } S \odot O) + (G_{\text{inpaint}} \odot (1 - O))$$
            \end{block}
            
            \vspace{-0.1cm}
            \begin{block}{Operational Pathways}
                \small
                \begin{itemize}
                    \item \textbf{Warping Pathway ($O \approx 1$):} Samples directly from source image $S$, preserving individual identity textures (dental braces).
                    \item \textbf{Inpainting Pathway ($O \approx 0$):} Activates under large deformation; synthesizes pixels using the VoxCeleb prior distribution (uniform white teeth).
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.44\textwidth}
            \begin{block}{Core Finding}
                \small
                \begin{itemize}
                    \item Pixel fidelity depends on the active pathway for each local region.
                    \item Regions with $O \approx 1$ preserve subject-specific details from $S$.
                    \item Regions with $O \approx 0$ reflect training dataset bias rather than the source.
                \end{itemize}
            \end{block}
