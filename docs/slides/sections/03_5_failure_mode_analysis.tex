\begin{frame}{Failure Mode Experiment: Dental Braces}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{block}{Experiment Setup}
                \small
                \begin{itemize}
                    \item \textbf{Test Idea:} How does FOMM handle rare facial features absent from training data?
                    \item \textbf{Subject:} Source portrait with metallic dental braces (\texttt{bella\_poarch\_dis\_teli}).
                    \item \textbf{Driving Motion:} Sequences from subtle head tilt to closed mouth, ending in a wide squinting smile.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Hypothesis}
                \small
                \begin{itemize}
                    \item VoxCeleb training data has virtually zero faces with dental braces.
                    \item \textbf{Warping Regime ($O \approx 1$):} Optical flow copies real source pixels $\to$ braces preserved.
                    \item \textbf{Inpainting Regime ($O \approx 0$):} Under extreme deformation, model flags teeth as occluded and synthesizes them.
                    \item \textbf{Prediction:} Inpainting will sample dataset average and erase braces.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- Slide 3: Root Cause Analysis & Video Links ---
\begin{frame}{Technical Mechanism: Dual-Path Synthesis}
    \begin{columns}[T]
        \begin{column}{0.52\textwidth}
            \begin{block}{Synthesis Formulation}
                \scriptsize
                $$\text{Output} = (\text{Warped } S \odot O) + (G_{\text{inpaint}} \odot (1 - O))$$
            \end{block}
            
            \vspace{-0.1cm}
            \begin{block}{Operational Pathways}
                \small
                \begin{itemize}
                    \item \textbf{Warping Path ($O \approx 1$):} Samples directly from source image $S$, preserving individual identity textures (dental braces).
                    \item \textbf{Inpainting Path ($O \approx 0$):} Activates under high deformation; synthesizes pixels using VoxCeleb feature distributions (uniform white teeth).
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.44\textwidth}
            \begin{block}{Core Finding}
                \small
                \begin{itemize}
                    \item Pixel fidelity depends on the active pathway for each region.
                    \item Regions with $O \approx 1$ mirror source specifics.
                    \item Regions with $O \approx 0$ reflect training dataset priors.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
