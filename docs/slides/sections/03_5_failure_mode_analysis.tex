% --- Slide 1: Hypothesis & Setup for Edge Case ---
\begin{frame}{Failure Mode Experiment: Dental Braces}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{block}{Experimental Setup}
                \small
                \begin{itemize}
                    \item \textbf{Goal:} Test FOMM on rare features absent from training data (OOD).
                    \item \textbf{Subject:} Source portrait with metallic dental braces.
                    \item \textbf{Motion:} Sequence from head tilt to lip closure and wide smile.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Hypothesis \& Expected Regimes}
                \small
                \begin{itemize}
                    \item VoxCeleb training data has no faces with braces.
                    \item \textbf{Warping ($O \approx 1$):} Samples source pixels $\to$ braces preserved.
                    \item \textbf{Inpainting ($O \approx 0$):} Large deformation masks mouth $\to$ model synthesizes generic teeth.
                    \item \textbf{Prediction:} Inpainting erases braces using dataset average.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- Slide 3: Root Cause Analysis & Video Links ---
\begin{frame}{Technical Mechanism: Dual-Path Synthesis}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{block}{Synthesis Formulation}
                \scriptsize
                $$\text{Output} = (\text{Warped } S \odot O) + (G_{\text{inpaint}} \odot (1 - O))$$
            \end{block}
            
            \vspace{-0.15cm}
            \begin{block}{Operational Pathways}
                \footnotesize
                \begin{itemize}
                    \item \textbf{Warping ($O \approx 1$):} Copies from source $S$, preserving identity details (braces).
                    \item \textbf{Inpainting ($O \approx 0$):} Under deformation, synthesizes pixels from VoxCeleb average (plain teeth).
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Core Finding}
                \footnotesize
                \begin{itemize}
                    \item Pixel fidelity depends on the active pathway.
                    \item $O \approx 1$ regions preserve source specifics.
                    \item $O \approx 0$ regions reflect training dataset bias.
                \end{itemize}
            \end{block}
