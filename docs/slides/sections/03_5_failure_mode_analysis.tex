\begin{frame}{Failure Mode Experiment: Dental Braces}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{block}{Experimental Setup}
                \small
                \begin{itemize}
                    \item \textbf{Goal:} Test FOMM on rare features absent from training data (OOD).
                    \item \textbf{Subject:} Source portrait with metallic dental braces.
                    \item \textbf{Motion:} Sequence from head tilt to lip closure and wide smile.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Hypothesis \& Expected Regimes}
                \small
                \begin{itemize}
                    \item VoxCeleb training data has no faces with braces.
                    \item \textbf{Warping ($O \approx 1$):} Samples source pixels $\to$ braces preserved.
                    \item \textbf{Inpainting ($O \approx 0$):} Large deformation masks mouth $\to$ model synthesizes generic teeth.
                    \item \textbf{Prediction:} Inpainting erases braces using dataset average.
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- Slide 3: Root Cause Analysis & Video Links ---
\begin{frame}{Technical Mechanism: Feature Warping \& Occlusion}
    \begin{columns}[T]
        \begin{column}{0.50\textwidth}
            \begin{block}{Synthesis Formulation}
                \scriptsize
                $$\text{Output} = D\big(W(E(S),\, T) \odot O\big)$$
                One decoder path: no inpainting branch, no RGB blend.
            \end{block}
            
            \vspace{-0.15cm}
            \begin{block}{Mechanism Steps}
                \footnotesize
                \begin{itemize}
                    \item \textbf{Encode:} the source $S$ becomes features $E(S)$.
                    \item \textbf{Warp:} the dense flow $T$ moves the features.
                    \item \textbf{Weight:} the mask $O$ multiplies the warped features.
                    \item \textbf{Decode:} $D(\cdot)$ renders the whole output frame.
                \end{itemize}
            \end{block}
        \end{column}

        \begin{column}{0.48\textwidth}
            \begin{block}{Core Finding}
                \footnotesize
                \begin{itemize}
                    \item At the mouth, the occlusion mask fails ($O \approx 0$).
                    \item $O \approx 1$ regions keep the warped source details.
                    \item $O \approx 0$ regions fall back on learned appearance priors.
                \end{itemize}
            \end{block}

                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
