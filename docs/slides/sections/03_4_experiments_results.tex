\begin{frame}[fragile]{Inference Pipeline: Python Pseudocode}
    \begin{columns}[T]
        \begin{column}{0.56\textwidth}
            \begin{block}{Frame-by-Frame Execution}
                \scriptsize
\begin{verbatim}
# 1. Reference keypoints from source portrait
kp_source = keypoint_detector(source_image)

# 2. Iterate over driving video
for frame_d in driving_video:
    # Blackbox 1: Motion tracking
    kp_driving = keypoint_detector(frame_d)

    # Blackbox 2: Dense motion estimation
    flow, occ = dense_motion_net(kp_source, kp_driving)

    # Blackbox 3: Warping & Inpainting
    out_frame = generator(source_image, flow, occ)

    yield out_frame
\end{verbatim}
            \end{block}
        \end{column}

        \begin{column}{0.41\textwidth}
            \textbf{Core Module Functions}
            \begin{itemize}
                \small
                \item \texttt{keypoint\_detector}: Extracts 10 self-supervised reference points.
                \item \texttt{dense\_motion\_net}: Converts sparse point shifts into optical flow and an occlusion mask.
                \item \texttt{generator}: Warps source pixels and inpaints missing areas (mouth/teeth).
            \end{itemize}

            \vspace{0.2cm}
            \begin{block}{Architectural Property}
                \scriptsize \textbf{No Recurrence:} The loop carries no hidden state across frames ($t \to t+1$).
            \end{block}
        \end{column}
    \end{columns}
\end{frame}
